60template<typename T>
61void hash_input(hash_type& seed, const sigma::Uncertain<T>& value) {
62 hash_input(seed, value.mean());
63 hash_input(seed, value.sd());
64 // deps() is an unordered container, so its iteration order is not stable
65 // (e.g. across copies of the same value). Fold each dependency into an
66 // order-independent (commutative) accumulator before combining so that two
67 // equal values always produce the same hash.
68 hash_type deps_hash = 0;
69 for(const auto& [dep, deriv] : value.deps()) {
70 hash_type term = 0;
71 hash_input(term, dep);
72 hash_input(term, deriv);
73 deps_hash ^= term;
74 }
75 hash_input(seed, deps_hash);
76}

84template<typename T>
85void hash_input(hash_type& seed, const types::affine_type<T>& value) {
86 hash_input(seed, value.center());
87 // error_terms() is an unordered_map, so its iteration order is not stable
88 // (e.g. across copies of the same value). Fold each error term into an
89 // order-independent (commutative) accumulator before combining so that two
90 // equal affine forms always produce the same hash.
91 hash_type terms_hash = 0;
92 for(const auto& [label, coeff] : value.error_terms()) {
93 hash_type term = 0;
94 hash_input(term, label);
95 hash_input(term, coeff);
96 terms_hash ^= term;
97 }
98 hash_input(seed, terms_hash);
99}

120template<typename T>
121void hash_input(hash_type& seed, const types::taylor_model_type<T>& value) {
122 // Unlike the other UQ types, TaylorModel has an explicit empty state
123 // (representing the empty set, distinct from a zero-width [0,0]
124 // remainder); remainder().lower()/upper() throw std::domain_error for an
125 // empty model, which would violate this function's no-throw guarantee.
126 if(value.empty()) {
127 hash_input(seed, std::string("empty_taylor_model"));
128 return;
129 }
130 hash_input(seed, value.constant());
131 // coefficients() is a std::map keyed on Monomial (which orders variables
132 // lexicographically), so iteration order is already stable -- unlike the
133 // unordered_map cases above, no order-independent folding is needed.
134 for(const auto& [mono, coeff] : value.coefficients()) {
135 for(const auto& [var, exponent] : mono.exponents()) {
136 hash_input(seed, var);
137 hash_input(seed, exponent);
138 }
139 hash_input(seed, coeff);
140 }
141 hash_input(seed, value.remainder().lower());
142 hash_input(seed, value.remainder().upper());
143}

147class HashVisitor {

154 void operator()(std::span<const T> data) {
155 for(std::size_t i = 0; i < data.size(); ++i) {
