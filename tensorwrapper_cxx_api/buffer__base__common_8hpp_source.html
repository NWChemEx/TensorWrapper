157 template<typename... Offsets>

180 template<typename... Args>

182 static_assert(sizeof...(Args) >= 1,

184 set_element_unpack_(std::make_index_sequence<sizeof...(Args) - 1>{},

200 template<typename OtherDerived>
201 bool operator==(const BufferBaseCommon<OtherDerived>& rhs) const noexcept {
202 if(has_layout() != rhs.has_layout()) return false;
203 if(has_layout() && layout().are_different(rhs.layout())) return false;
204 return true;
205 }

217 return !(*this == rhs);

227 template<typename OtherDerived>

231 }
232
233protected:
234 void assert_layout_() const {
235 if(!has_layout()) {
236 throw std::runtime_error(
237 "Buffer has no layout. Was it default initialized?");
238 }
239 }

241private:

250 template<std::size_t... Is, typename Tuple>
251 void set_element_unpack_(std::index_sequence<Is...>, Tuple&& values) {
252 set_element(
253 index_vector{static_cast<size_type>(std::get<Is>(values))...},
254 static_cast<element_type>(std::get<sizeof...(Is)>(values)));
255 }

262 }
